// 🧠 What is flatMap?

// 👉 flatMap = map + flatten

// map = change each item

// flatten = remove nested arrays

// 🧩 Simple Example
// Normal Array
// const numbers = [1, 2, 3];
// ✅ map (only changes items)
// const result = numbers.map(n => [n, n * 2]);

// Result:

// [[1,2], [2,4], [3,6]]

// 👉 Notice: array inside array (nested)

// ✅ flatMap (map + flatten)
// const result = numbers.flatMap(n => [n, n * 2]);

// Result:

// [1, 2, 2, 4, 3, 6]

// 👉 flatMap removes nested arrays automatically.
